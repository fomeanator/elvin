scene expand_loops

// РАЗВОРОТЫ ДО РАЗБОРА. Цикл и вызов функции — не команды, а ПЕРЕСТАНОВКА
// СТРОК: к моменту разбора их уже нет. Отсюда и цена ошибки — цикл, свёрнутый
// на шаг короче, компилируется молча и играется неправильно.
//
// Свидетель на то, что переезд в LvnsCompiler.Expand ничего не сдвинул: имена
// синтетических меток попадают в сейв игрока (якорь сейва — ближайшая метка),
// и сместить их значило бы увести чужую закладку.

func mine(ore, rate) {
    gold = gold + ore * rate
    __ret = gold
}

func report(place) {
    Кладовщик: На складе {place} намыто {gold}.
}

:start
gold = 0
ores = [1, 2, 3]

for ore in ores {
    mine(ore, 2)
}

// Вызов с присваиванием результата: связать доводы, прыгнуть, забрать __ret.
total = mine(5, 3)

// Однострочная форма: тело разворачивается в собственные строки.
for ore in ores { gold = gold + ore }

:deep
left = 2
while left > 0 {
    for ore in ores {
        if ore > 2 {
            Рассказчик: Тяжёлая ходка.
        } else {
            Рассказчик: Лёгкая ходка.
        }
    }
    left = left - 1
}

report("южный")
Рассказчик: Смена закрыта, итог {total}.
