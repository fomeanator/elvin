scene func_return

# Функция, тело которой КОНЧАЕТСЯ на return, и функция, которая на нём не
# кончается. Страховочный `return` ставится только второй: первой он достался
# бы вторым подряд, и второй недостижим — валидатор о таком сообщает.
func stop_here(a) {
    x = a
    return
}

func falls_through(b) {
    y = b
}

:start
stop_here(1)
falls_through(2)
Кладовщик: Готово, {x} и {y}.
